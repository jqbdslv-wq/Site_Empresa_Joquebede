-- ============================================================
-- Configuração do Supabase - Site de Cadastro de Funcionários
-- Cole este script no SQL Editor do Supabase e clique em "Run"
-- ============================================================

-- 1) Tabela de funcionários
create table if not exists public.funcionarios (
  id           bigint generated by default as identity primary key,
  nome         text not null,
  email        text not null,
  cargo        text,
  departamento text,
  foto_url     text,
  created_at   timestamptz not null default now()
);

-- 2) RLS (Row Level Security) ligado
alter table public.funcionarios enable row level security;

-- 3) Políticas de acesso (site público, sem login)
drop policy if exists "leitura publica" on public.funcionarios;
create policy "leitura publica" on public.funcionarios
  for select using (true);

drop policy if exists "insert publico" on public.funcionarios;
create policy "insert publico" on public.funcionarios
  for insert with check (true);

-- 4) Bucket público para as fotos dos funcionários
insert into storage.buckets (id, name, public)
values ('fotos-funcionarios', 'fotos-funcionarios', true)
on conflict (id) do nothing;

drop policy if exists "upload de fotos" on storage.objects;
create policy "upload de fotos" on storage.objects
  for insert to anon, authenticated
  with check (bucket_id = 'fotos-funcionarios');

drop policy if exists "leitura de fotos" on storage.objects;
create policy "leitura de fotos" on storage.objects
  for select to anon, authenticated
  using (bucket_id = 'fotos-funcionarios');
